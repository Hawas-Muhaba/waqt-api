# Sprint 11: waqt-core (astronomy, Hijri calendars, scheduling,
# institutional calendars) is now an independently distributable
# package under core/, and this repo's root package (waqt.api) depends
# on it formally (see pyproject.toml). Installing both in editable
# mode reproduces the old "just works from a checkout" experience
# while making the dependency real and testable -- `pip install -e
# ./core` alone pulls in waqt-core's own dependencies (ephem,
# timezonefinder), and `pip install -e .` pulls in fastapi/uvicorn per
# the root pyproject.toml.
-e ./core
-e .

# waqt_local (local/) similarly depends on waqt-core, not on this
# repo's root package -- see local/pyproject.toml.
-e ./local
